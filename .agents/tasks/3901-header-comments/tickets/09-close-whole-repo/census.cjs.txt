const ts = require(process.cwd() + '/node_modules/typescript');
const fs = require('fs');

const files = fs.readFileSync(process.argv[2], 'utf8').split('\n')
  .filter(f => /\.(ts|tsx|js|mjs|cjs|mts)$/.test(f) && !f.startsWith('.agents/tasks/'));
const DIRECTIVE = /^\s*(\/\/|\/\*+)\s*(eslint|SPDX|@vitest-environment|@ts-|istanbul|prettier|c8|Stryker|@jest|global |globals |cspell|webpack|@__PURE__|#__PURE__)/;

function area(f) {
  const p = f.split('/');
  if (p[0] === 'src') {
    if ([ 'features', 'slices', 'services', 'shared', 'toolkit' ].includes(p[1])) return p.slice(0, 3).join('/');
    return p.slice(0, 2).join('/');
  }
  if ([ 'tools', 'deploy', '.agents' ].includes(p[0])) return p.slice(0, 2).join('/');
  return p.length > 1 ? p[0] : '(root)';
}

const stats = {};
const rows = [];
for (const f of files) {
  const text = fs.readFileSync(f, 'utf8');
  const kind = /\.(tsx|js|mjs|cjs)$/.test(f) ? ts.ScriptKind.TSX : ts.ScriptKind.TS;
  const sf = ts.createSourceFile(f, text, ts.ScriptTarget.Latest, true, kind);
  const comments = [];
  const seen = new Set();
  function collect(pos) {
    for (const fn of [ ts.getLeadingCommentRanges, ts.getTrailingCommentRanges ]) {
      for (const c of fn(text, pos) || []) {
        if (!seen.has(c.pos)) { seen.add(c.pos); comments.push(c); }
      }
    }
  }
  function visit(node) {
    collect(node.pos);
    collect(node.end);
    ts.forEachChild(node, visit);
  }
  visit(sf);
  comments.sort((a, b) => a.pos - b.pos);
  const line = p => sf.getLineAndCharacterOfPosition(p).line;
  const firstTokenLine = sf.statements.length ? line(sf.statements[0].getStart(sf)) : Infinity;
  const blocks = [];
  for (const c of comments) {
    const t = text.slice(c.pos, c.end);
    const sl = line(c.pos);
    const el = line(c.end);
    const last = blocks[blocks.length - 1];
    const ownLine = text.slice(text.lastIndexOf('\n', c.pos - 1) + 1, c.pos).trim() === '';
    const single = c.kind === ts.SyntaxKind.SingleLineCommentTrivia;
    if (last && single && last.single && ownLine && last.ownLine && sl === last.el + 1) {
      last.el = el; last.text += '\n' + t; continue;
    }
    blocks.push({ sl, el, text: t, single, ownLine });
  }
  const a = area(f);
  const s = stats[a] ||= { files: new Set(), top: 0, jsdoc: 0, over8: 0, inscope: 0, lines: 0 };
  let seenNonDirective = false;
  for (const b of blocks) {
    const n = b.el - b.sl + 1;
    if (DIRECTIVE.test(b.text) && n === 1) continue;
    const jsdoc = b.text.startsWith('/**') && !b.text.startsWith('/***');
    if (jsdoc && n === 1 && /@type/.test(b.text)) continue;
    const top = b.sl < firstTokenLine && !seenNonDirective;
    seenNonDirective = true;
    if (!(top || n >= 3 || jsdoc)) continue;
    s.files.add(f); s.inscope++; s.lines += n;
    if (top) s.top++;
    if (jsdoc) s.jsdoc++;
    if (n > 8) s.over8++;
    rows.push([ f, b.sl + 1, n, top ? 'top' : '', jsdoc ? 'jsdoc' : '' ].join('\t'));
  }
}
fs.writeFileSync(process.argv[3], rows.join('\n'));
const out = Object.entries(stats).filter(([ , s ]) => s.inscope).sort((a, b) => b[1].lines - a[1].lines);
const T = { files: 0, blocks: 0, lines: 0, jsdoc: 0, over8: 0 };
console.log('area\tfiles\tblocks\tlines\ttop\tjsdoc\tover8');
for (const [ a, s ] of out) {
  console.log([ a, s.files.size, s.inscope, s.lines, s.top, s.jsdoc, s.over8 ].join('\t'));
  T.files += s.files.size; T.blocks += s.inscope; T.lines += s.lines; T.jsdoc += s.jsdoc; T.over8 += s.over8;
}
console.log('TOTAL', JSON.stringify(T));
